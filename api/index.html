def enrich(raw_docstring: str, signature: Signature) -> dict[str, object]:
    """
    title: Return structured documentation enriched with
      annotation types.
    summary: Types come exclusively from annotations. No
      objects or source files are modified. Missing annotations
      raise DocumentationError.
    parameters:
      raw_docstring: >-
        Authored Douki YAML containing descriptions and no
        type fields.
      signature: Annotation metadata supplied by the language
        adapter.
    returns: >-
      New documentation mapping with authored descriptions
      and annotation-derived types.
    """
    try:
        data = normalize_documentation(
            cast(
                dict[str, object],
                _parse_validated_docstring(inspect.cleandoc(raw_docstring)),
            )
        )
    except ValueError as error:
        raise DocumentationError('schema', str(error)) from error
    validate_annotations(signature, data)
    if signature.parameters is not None:
        parameters = merge_fields(
            cast(dict[str, object], data.get('parameters', {})),
            signature.parameters,
            strict=True,
        )
        if parameters or 'parameters' in data:
            data['parameters'] = parameters
    if signature.attributes:
        data['attributes'] = merge_fields(
            cast(dict[str, object], data.get('attributes', {})),
            signature.attributes,
            strict=False,
        )
    if signature.returns:
        returns = entry(data.get('returns'))
        returns['type'] = signature.returns
        data['returns'] = returns
    return data

@typechecked
def format_documentation(
    data: dict[str, object],
    *,
    width: int = 79,
    enriched: bool = False,
) -> str:
    """
    title: Serialize validated source or explicitly enriched documentation.
    summary: Only generated output may contain annotation-derived type fields.
    parameters:
      data: Documentation mapping to validate and serialize.
      width: Target line width for the serialized YAML.
      enriched: >-
        Whether to allow annotation-derived type fields in generated output.
    returns: Serialized YAML with canonical field order and preserved values.
    """
    try:
        validate_schema(data, enriched=enriched)
    except ValueError as error:
        raise DocumentationError('schema', str(error)) from error
    ordered = {key: data[key] for key in _KEY_ORDER if key in data}
    target = max(20, width)
    wrap_width = target
    for _ in range(4):
        text = yaml.dump(
            ordered,
            Dumper=DocumentationDumper,
            sort_keys=False,
            allow_unicode=True,
            width=wrap_width,
        )
        # PyYAML treats width as a hint and can overshoot by one word.
        excess = max(
            (len(line) - target for line in text.splitlines()), default=0
        )
        if excess <= 0 or wrap_width == 20:
            break
        wrap_width = max(20, wrap_width - max(excess, 8))
    return text

    def get_version() -> str:
    """
    title: Return the program version.
    returns: >-
      Installed package version, or the source version when not installed.
    """
    try:
        return importlib_metadata.version(__name__)
    except importlib_metadata.PackageNotFoundError:  # pragma: no cover
        return '0.14.0'  # semantic-release

@typechecked
def getdoc(obj: object, *, skip_receiver: bool = False) -> str | None:
    """
    title: Return enriched Douki YAML or an ordinary cleaned docstring.
    summary: The object's original __doc__ is never changed.
    parameters:
      obj: Declaration whose documentation or annotation metadata is needed.
      skip_receiver: Whether to omit the first parameter of an unbound method.
    returns: >-
      Enriched Douki YAML, cleaned ordinary prose, or None when no docstring
      exists.
    """
    alternatives = variants(obj)
    if len(alternatives) > 1 or alternatives[0] is not obj:
        documents = [
            variant_doc(variant, obj, skip_receiver)
            for variant in alternatives
        ]
        return '\n---\n'.join(doc for doc in documents if doc) or None
    raw = inspect.getdoc(obj)
    if raw is None:
        return None
    # Ordinary prose remains ordinary prose. A title key opts into Douki.
    if not is_douki(raw):
        return raw
    return format_documentation(
        enrich(raw, metadata(obj, skip_receiver)), enriched=True
    )

    @public
@typechecked
def help(request: object = no_request) -> None:
    """
    title: Route registered objects to Douki and other requests to Python help.
    summary: No-argument interactive help and string topics use Python help.
    parameters:
      request: >-
        Object or topic to document; omitting it opens interactive Python help.
    """
    if request is no_request:
        fallback_help()
    elif isinstance(request, str):
        fallback_help(request)
    elif contains_douki(request, set()):
        name = getattr(request, '__name__', type(request).__name__)
        pydoc.pager('Help on ' + Renderer().visit(request, name) + '\n')
    else:
        fallback_help(request)

    @public
@typechecked
def register(package: str) -> None:
    """
    title: Register a library for annotation-enriched help.
    summary: Call once from the library's __init__.py using __name__.
      Registration includes future submodules without importing them.
      Other packages continue to use the previously installed help
      helper.
    parameters:
      package: Full module or package import name to register.
    raises:
      DocumentationError: The name is empty or is not a dotted identifier.
    """
    global fallback_help
    if not all(part.isidentifier() for part in package.split('.')):
        raise DocumentationError('registration', 'Expected a dotted name')
    # Install once. A later helper may delegate back to ours, so capturing
    # it again would create a recursion cycle or overwrite another hook.
    if not registered_packages and builtins.help is not help:
        fallback_help = builtins.help
        setattr(builtins, 'help', help)
    registered_packages.add(package)

    @public
@typechecked
def validate_annotations(
    signature: Signature,
    documentation: dict[str, object] | None = None,
) -> None:
    """
    title: Require explicit annotations at the shared documentation boundary.
    summary: Adapters omit implicit method receivers from parameter metadata.
      Documented attributes must have declarations; values are never used
      to guess their types.
    parameters:
      signature: Annotation metadata supplied by the language adapter.
      documentation: >-
        Authored mapping used to check that documented attributes are declared.
    """
    missing: list[str] = []
    if signature.parameters is not None:
        missing.extend(
            f"parameter '{parameter.name}'"
            for parameter in signature.parameters
            if not parameter.annotation.strip().strip('\'"').strip()
        )
        if not signature.returns.strip().strip('\'"').strip():
            missing.append('return value (use -> None for no result)')
    attributes = {item.name: item.annotation for item in signature.attributes}
    documented = (documentation or {}).get('attributes', {})
    if isinstance(documented, dict):
        for name in documented:
            attributes.setdefault(name, '')
    missing.extend(
        f"attribute '{name}'"
        for name, annotation in attributes.items()
        if not annotation.strip().strip('\'"').strip()
    )
    if missing:
        raise DocumentationError(
            'annotations',
            'Missing type annotations: ' + ', '.join(missing),
        )
